dat$Date <- dmy(dat$Date)
dat$week <- lubridate::isoweek(dat$Date)
dat_mod <- dat[dat$`Functioning Day` == "Yes",]
dat_mod <- dat_mod |> 
  dplyr::rename("temp"="Temperature(°C)" ,
                "rain"="Rainfall(mm)",
                "snow"="Snowfall (cm)")
X <- model.matrix(~ as.factor(Hour) + temp + snow + rain + as.factor(Holiday),dat_mod)[,-1]
ctr_X <- scale(X, scale = FALSE)
qr_X <- qr(ctr_X)
n_col <- ncol(X)
n_row <- nrow(X)
Q_s <- qr.Q(qr_X) %*% diag(rep(sqrt(n_row-1),n_col))
R_inv_D <- solve(qr.R(qr_X)) %*% diag(rep(sqrt(n_row-1),n_col))
stan_data <- list(
  N = n_row,
  p = n_col,
  Q = Q_s,
  y = dat_mod$`Rented Bike Count`,
  R_inv_D = R_inv_D,
  W = length(unique(dat_mod$week)),
  wk_idx = dat_mod$week
)
data {
    int<lower=0> N;
    int<lower=0> p;
    int<lower=0> W;
    // observations
    matrix[N, p] Q;
    matrix[p, p] R_inv_D;
    array[N] int<lower=0> y;
    array[N] int<lower=0, upper=W> wk_idx;
}
parameters {
  real alpha;
  vector[W] alpha_week;
  real<lower=0> tau_week;
  vector[p] theta;
  real<lower=0> phi;
}
model {
  alpha_week ~ normal(alpha, tau_week);
  y ~ neg_binomial_2_log_glm(Q, alpha_week[wk_idx], theta, phi);
}
generated quantities {
  vector[p] beta = R_inv_D * theta;
  array[N] int y_rep = neg_binomial_2_log_rng(alpha_week[wk_idx] + Q * theta, phi);
}
fit_bikes <- bikes$sample(stan_data, parallel_chains = 4)
fit_bikes$summary("beta")
y <- stan_data$y
y_rep <- fit_bikes$draws("y_rep", format = "draws_matrix")
pp_check(stan_data$y, y_rep[1:50,], ppc_dens_overlay)
pp_check(stan_data$y, y_rep[1:50,], ppc_rootogram)
ppc_stat_2d(y, yrep = y_rep, stat = c("min","max"))
